## 一、生成器

### 1. 生成器表达式

```python
# 可以通过next()方法或for循环来获取生成器的值
# 定义生成器
t1=(i*2 for i in range(6))
# next()获取生成器的值
print(next(t1))
print(next(t1))
print(next(t1))
print(next(t1))
print(next(t1))
print(next(t1))
# for循环获取生成器的值
for i in t1:
    print(i)
```

### 2. 函数生成器

```python
# 灵活的生成器
# 1.是一个函数
# 2.函数内部有yield关键字,函数的结果就是一个生成器
def fn():
    for i in range(5):
        print('函数开始')
        yield i**2
        print('函数结束')
# 普通函数的执行结果：函数的返回值
# 生成器函数的执行结果：yield的返回值
generator=fn() # 获取生成器，再靠生成器获取结果
# 可以通过next()获取生成器的结果
print(next(generator))
print(next(generator))
print(next(generator))
print(next(generator))
print(next(generator))
# 也可以通过for循环获取生成器的结果
for i in generator:
    print(i)
```

### 3. 协程

```python
# await不能单独使用，必须搭配async
# 创建多个协程任务asyncio.create_task需要一个入口函数
# asyncio.run执行入口函数
# 创建任务传入的是函数的调用，并发的是任务，开启任务传入的是函数的调用
import asyncio
# 1.定义协程函数
async def eat():
    print('开始吃了')
    # 2.让出CPU控制权
    # 3.等待
    await asyncio.sleep(2)
    print('吃饱了')
async def drink():
    print('开始喝了')
    await asyncio.sleep(1)
    print('喝饱了')
# 4.入口函数
async def main():
    # 5.创建协程任务
    task1=asyncio.create_task(eat())
    task2=asyncio.create_task(drink())
    # 6.并发执行协程任务
    await task1
    await task2
# 7.执行入口任务
asyncio.run(main())
```

## 二、正则表达式

### 1. 匹配方式

```python
import re
# match方法：从头匹配
# 匹配成功返回对象，匹配失败返回None
r1=re.match('it','itheima')
r2=re.match('hei','itheima')
print(r1,r2)
# search方法：从任意位置匹配，查找是否包含
# 匹配成功返回对象，匹配失败返回None
r3=re.search('th','itheima')
r4=re.search('hei','itheima')
print(r3,r4)
# findall方法：从任意位置匹配，查找是否包含，返回所有匹配的字符串
# 返回的是列表
r5=re.findall('s','itheima')
r6=re.findall('i','itheima')
print(r5,r6)
```

### 2. 单个字符匹配

```python
# .匹配任意1个字符
r1=re.match(r'.','itheima')
# [] 匹配指定字符
r2=re.match(r'[a-z]','itheima')
# [^指定字符] 匹配非指定字符
r3=re.match(r'[^a-zA-Z]','itheima')
# \d匹配数字
r4=re.match(r'\d','itheima')
# \D匹配非数字
r5=re.match(r'\D','itheima')
# \s匹配空格
r6=re.match(r'\s','itheima')
# \S匹配非空格
r7=re.match(r'\S','itheima')
# \w匹配数字、字母、下划线
r8=re.match(r'\w','itheima')
# \W匹配特殊字符
r9=re.match(r'\W','itheima')
print(r1,r2,r3,r4,r5,r6,r7,r8,r9)
```

### 3. 多个字符匹配

```python
import re
# * 匹配前一个字符出现0次或者多次
r1=re.match(r'.*','itheima')
# + 匹配前一个字符出现1次或者多次
r2=re.match(r'.+','itheima')
# ? 匹配前1个字符出现0次或者1次
r3=re.match(r'\d?','itheima')
# {m}匹配前一个字符出现m次
r4=re.match(r'.{8}','itheima')
# {m,n}匹配前一个字符出现m到n次
r5=re.match(r'\w{1,7}','itheima')
```

### 4. 开头结尾匹配

```python
import re
# 11位手机号
r1=re.match(r'^1[3-9]\d{9}$','13892380711')
print(r1)
# 6位纯数字验证码
r2=re.match(r'^\d{6}$','123456')
print(r2)
```

### 5. 贪婪与非贪婪

```python
import re
# 贪婪匹配：匹配尽可能多的内容
content='<img src="http://www.baidu.com/a.png" width="100%">'
r1=re.findall(r'^<img src="(.*)".*>$',content)
print(r1)
# 非贪婪匹配：匹配尽可能少的内容
r2=re.findall(r'^<img src="(.*?)".*>$',content)
print(r2)
```

### 6. 分组

```python
import re
# | 匹配任意一个
r=re.match(r'^[a-zA-Z0-9]{4,12}@(qq|163|sina)\.com$','lizhengyang@qq.com')
print(r)
# 分组（）  \第几组
content1='<h1>李正洋</h1>'
content2='<h2>李正洋</h2>'
content3='<h1>李正洋</h2>'
r1=re.match(r'^<[a-z0-9]{2}>.*</[a-z0-9]{2}>$',content1)
r2=re.match(r'^<[a-z0-9]{2}>.*</[a-z0-9]{2}>$',content2)
r3=re.match(r'^<([a-z0-9]{2})>.*</\1>$',content1)
r4=re.match(r'^<([a-z0-9]{2})>.*</\1>$',content2)
r5=re.match(r'^<([a-z0-9]{2})>.*</\1>$',content3)
print(r1,r2,r3,r4,r5)
```
