## DML综合案例

需求说明：在DDL表的基础上实施

```properties
-- DML 综合案例
# 插入数据
# # 插入用户数据：
# 1, 'zhangsan', '123456', 'zhangsan@example.com', '13800138000', '张三'
# 2, 'lisi', '123456', 'lisi@example.com', '13900139000', '李四'
# 3, 'wangwu', '123456', 'wangwu@example.com', '13700137000', '王五'

insert into shop_db.user values
                                (1, 'zhangsan', '123456', 'zhangsan@example.com', '13800138000', '张三'),
                                (2, 'lisi', '123456', 'lisi@example.com', '13900139000', '李四'),
                                (3, 'wangwu', '123456', 'wangwu@example.com', '13700137000', '王五');

# # 插入商品分类数据
# 'c01', '手机'
# 'c02', '家用电器'
# 'c03', '电脑'

-- 我的表id是int类型, 但是数据是字符串类型, 所以我需要修改表的字段类型
# alter table shop_db.category change category_id  category_id varchar(20);
alter table shop_db.category modify  category_id varchar(20);

insert into shop_db.category values ('c01', '手机'),('c02', '家用电器'),('c03', '电脑');

# # 插入商品数据
# 1, 'iphone16 Pro Max', 12999.00, 'c01'
# 2, '三星M25', 15999.00, 'c01'
# 3, '海尔冰箱', 4999.00, 'c02'
# 4, '联想电脑Y9000', 4399.90, 'c03'
# 5, '微波炉', 899.00, 'c02'

alter table shop_db.product modify category_id varchar(20);

insert into shop_db.product values
                                   (1, 'iphone16 Pro Max', 12999.00, 'c01'),
                                   (2, '三星M25', 15999.00, 'c01'),
                                   (3, '海尔冰箱', 4999.00, 'c02'),
                                   (4, '联想电脑Y9000', 4399.90, 'c03'),
                                   (5, '微波炉', 899.00, 'c02');

# # 插入订单数据
# 1001, 1, 12999.00, '已支付'
# 1002, 1, 4999.00, '已支付'
# 1003, 2, 15999.00, '待支付'
# 1004, 3, 899.00, '已取消'

insert into shop_db.order_info values
                                      (1001, 1, 12999.00, '已支付'),
                                      (1002, 1, 4999.00, '已支付'),
                                      (1003, 2, 15999.00, '待支付'),
                                      (1004, 3, 899.00, '已取消');

# # 更新数据
# 更新用户真实姓名：将user_id为1的用户的real_name改为'张老三'
update shop_db.user set  real_name = '张老三' where user_id = 1;
# 更新商品价格：将商品表中商品名称为'联想电脑Y9000'的价格改为4000.00
update shop_db.product set price = 4000.00 where product_name = '联想电脑Y9000';

# 更新订单状态：将订单状态为'待支付'的订单状态更新为'已支付'
update  shop_db.order_info set status = '已支付' where  status ='待支付';
#
# # 删除数据
# 删除用户：删除用户名为'wangwu'的用户
delete  from shop_db.user where username = 'wangwu';
# 删除订单：删除订单状态为'已取消'的订单
delete from shop_db.order_info where status = '已取消';

# 清空订单表
truncate table shop_db.order_info;
#delete  from shop_db.order_info;
```
