## 函数进阶

### 1. 函数的说明文档

~~~python
def get_sum(x,y):
    """
    这是一个求和函数
    :param x: 求和的第一个数
    :param y: 求和的第二个数
    :return: 返回的结果
    """
    return x+y
get_sum(1,2)
#函数说明文档是类似于注释的良好习惯，对于复杂的函数一定要进行说明
~~~

### 2. 嵌套调用

```python
# 生成一个长度10的随机列表并排序
import random
# 定义随机列表
def get_random_list(x,y):
    # 随机列表生成推导式
    """
    :param x: 范围x
    :param y: 范围y
    :return: 返回随机数列
    """
    random_list=[random.randint(x,y) for i in range(10)]
    return  random_list
# 定义排序函数
def sort_random_list(l1:list):
    # 按照从大到小排序
    l1.sort(reverse= True)
    print(l1)
# 定义获取排序完成的列表
def get_sort_random_list(x,y):
    l1=get_random_list(x,y)
    new_l1=sort_random_list(l1)
    return new_l1
get_sort_random_list(1,100)
```

### 3. 变量作用域

- 全局作用域
  - 全局变量
    - 在全局作用域定义的变量
- 局部作用域
  - 局部变量
    - 在函数内部定义的变量
- 全局作用域无法访问局部变量，全局变量可以在任何部位访问
- 函数执行完成后，局部变量会被回收
- 局部修改全局变量可以使用global关键字
- 如果函数外部要访问局部变量，只能通过return返回

```python
num=100 # 定义全局变量
def fn():
    num=10 # 定义局部变量
    print(num)  --10  # 函数执行完成后，局部变量消失
fn()
print(num)  --100  # 无法访问局部变量

# 修改全局变量
def fn():
    global num # 修改全局变量
    num=10
    age=18
    print(num)
    return age # 如果函数外部要访问局部变量，只能通过return返回
age=fn() # 变量接收局部变量返回值
print(num)
print(age)
```

### 4. 多返回值

- return x,y
- x,y=调用函数()

```python
def fn():
    num=10
    age=18
    return num,age
num,age=fn()

```

### 5. 多种传参方式

```python
# 1. 位置参数
# 形参和实参的位置和个数要一致
def fn(x,y):
    return x,y
total=fn(1,2)
print(total)
# 2. 关键字参数
# 通过key=value进行传参，可忽略位置，可以和位置参数混合使用
def get_info(name,age,gender):
    print(f'姓名{name},年龄{age},性别{gender}')
get_info(name='张三',gender='男',age=20)
get_info('李四',gender='男',age=20)
# 3. 缺省参数-默认值参数
# 不传递默认参数，以默认值为主，如果传递了参数，以传入参数为主
def get_info(name,age,gender='男'):
    print(f'姓名{name},年龄{age},性别{gender}')
get_info(name='张三',age=20)
# 4.不定长参数
# 4.1 位置传递
def get_sum(*args):  # 以元组形式接收参数
    total=sum(args)
    print(total)
get_sum(1,2,3,4,5)
# 4.2 关键字传递
def get_total(**kwargs):  # 以字典形式接收参数
    total=sum(kwargs.values())
    print(total)
get_total(x=5,y=8,z=6)
```

### 6. 匿名函数

- 如果函数比较简单，一行代码就可以实现，可以定义lambda表达式解决问题
- 语法：lambda 参数1，参数2：表达式

```python
# 定义匿名函数
fn=lambda x,y:x+y
# 传参
total=fn(1,2)  # lambda默认有return,所以能拿到结果
print(total)
# 简写
print(fn(1,2))

# 排序列表
stu = [
    {'name': 'Tom', 'age': 20},
    {'name': 'Rose', 'age': 19},
    {'name': 'Jack', 'age': 22}
]
res = sorted(stu, key=lambda params: params['age'], reverse=True)
print(res)
```

### 7.递归函数

- 函数内部调用自身
- 要反复使用函数，但不确定使用次数
- 三要素：
  - 把复杂内容拆分成相同的多件事
  - 找到出口
  - return返回值		

```python
# 递归 函数内部调用自身
# 死递归
def fn():
    print('fn执行')
    fn()
fn()
# 递归示例
count=1
def fn():
    print('fn执行')
    global count
    count+=1
    if count<=10: # 递归的出口
        fn()
fn()
# 求10之前所有正整数的和
def get_sum(n):
    if n==1:  # 递归的出口
        return 1
    return n+get_sum(n-1) # 递归的结果
print(get_sum(10))
# 斐波那契数列
def get_fb(n):
    if n==1:
        return 1
    if n==2:
        return 1
    return get_fb(n-1)+get_fb(n-2)
print(get_fb(9))
```
