代码示例：

```sql
设计一个电商系统数据库（shop_db），包含以下核心功能：
1. 用户管理
2. 商品分类管理
3. 商品信息管理
4. 订单管理

具体需求：
-- 综合案例:
-- DDL的综合案例
-- 1. 创建数据库并使用: shop_db
drop database if exists shop_db;
create database shop_db;
use shop_db;
-- 2. 创建用户表(user): user_id  username password email phone
drop table if exists shop_db.user;
create table shop_db.user(
  user_id int primary key ,
  username varchar(20),
  password varchar(20),
  email varchar(50),
  phone varchar(20)
);

-- 3. 创建商品分类表(category): category_id cname description
drop table if exists shop_db.category;
create table shop_db.category(
  category_id int primary key ,
  cname varchar(20),
  description varchar(100)
);

-- 4. 创建商品表(product):  product_id   product_name price category_id
drop table if exists shop_db.product;
create table shop_db.product(
  product_id int primary key ,
  product_name varchar(20),
  price decimal(10,2),
  category_id int
);

-- 5. 创建订单表(order): order_id  user_id total_amount status
drop table if exists shop_db.order;
create table shop_db.order(
  order_id int primary key ,
  user_id int,
  total_amount decimal(10,2),
  status varchar(20)
);

-- 6. 查看所有表
show tables ;
-- 7. 查看商品表结构
desc shop_db.product;
-- 8. 修改表结构操作
-- 8.1 为用户表添加真实姓名列 real_name
alter table shop_db.user add real_name varchar(20);
-- 8.2 删除分类表的描述列
alter table shop_db.category drop description;
-- 8.3 重命名订单表 order 重命名为  order_info
rename table shop_db.order to shop_db.order_info;
-- 9. 最终查看所有表
show tables ;
```
