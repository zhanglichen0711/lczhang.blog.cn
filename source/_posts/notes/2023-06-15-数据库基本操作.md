## MySQL数据库

## 一、DDL

```python
# DDL：数据库的新增、删除、修改

# 1.创建数据库
create database day06_db;
# 没有了才创建，有了也不报错
# create database if not exists day06_db;
# 字符集设置
# create database if not exists day06_db character set utf8;

# 2.查看数据库
show databases;

# 3.切换（使用）数据库
use day06_db;

# 4.删除数据库
drop database day06_db;
drop database if exists day06_db;

# 5.创建表
create table student(
    id int primary key auto_increment,  # 主键：唯一、非空
    name varchar(20) not null, # 非空
    sid int unique,   # 唯一
    create_time datetime default current_timestamp(),  # 默认约束
    height decimal(5,2)
);
# 数据类型：int 整数，decimal(5,2) 小数，varchar(20) 字符串，datetime 时间
# 数据约束：primary key 主键，not null 非空，unique 唯一，default 默认
create table shop_db(
    id int primary key auto_increment,
    name varchar(20) not null,
    price decimal(10,2),
    product_time datetime,
    category varchar(20)
);

# 6.删除表
drop table student;
# drop table day06_db.shop_db;

# 7.查看当前数据库的表
show tables;

# 8.查看表结构
desc shop_db;
```

## 二、DML

```python
# DML：数据的新增、删除、修改
# 1.新增数据
# 1.1 按照字段插入对应的值
insert into shop_db(name,price,product_time,category) values('苹果',8.8,'2026-06-20','水果');
# 1.2 省略字段
insert into day06_db.shop_db values(1,'苹果',8.8,'2026-06-20','水果');
# 1.3 批量插入
insert into day06_db.shop_db values
                                 (1,'苹果',8.8,'2026-06-20','水果'),
                                 (2,'香蕉',5.5,'2026-06-20','水果'),
                                 (3,'橘子',6.6,'2026-06-20','水果');
# 2.修改数据
update day06_db.shop_db set name='西瓜' where id=2; # 修改指定数据
update day06_db.shop_db set price=3.0; # 修改整列
# 3.删除数据
delete from day06_db.shop_db where id=3; # 删除指定行
delete from day06_db.shop_db; # 删除整表（记录还在）
truncate table day06_db.shop_db; # 删除整表（恢复出厂设置）
```

## 三、DQL

```python
# DQL：数据查询

# 1.简单查询
# select 列1(as 别名1),列2(as 别名2) from 库.表名 as 别名
# 查询所有列
select * from day06_db.shop_db;
# 查询指定列
select name,price from day06_db.shop_db;
select name as '商品名称',price as '商品价格' from day06_db.shop_db;

# 2.条件查询
# select 列1(as 别名1),列2(as 别名2) from 库.表名 as 别名 where 条件

# 2.1 比较运算符 = < > <= >= !=
select * from day06_db.shop_db where name='苹果';

# 2.2 逻辑运算符 and并且 or或者 not非
select * from day06_db.shop_db where name='苹果' or name='西瓜';

# 2.3 范围查询 between...and...连续范围查询  in()非连续范围查询
select * from day06_db.shop_db where price between 1 and 5;
select * from day06_db.shop_db where price in(1,3,5);

# 2.4 模糊查询 like % 任意多个字符 _ 任意一个字符
select * from day06_db.shop_db where name like '%果%';
select * from day06_db.shop_db where name like '_果%';

# 2.5 空查询 is null 空值 is not null 非空
select * from day06_db.shop_db where price is not null;

# 3.排序查询 order by 列  asc升序 desc降序
select * from day06_db.shop_db order by name;
select * from day06_db.shop_db order by name desc;

# 4.聚合函数 多进一出 count sum avg max min
select count(*) as '商品数量' from day06_db.shop_db;
select sum(price) as '商品总价' from day06_db.shop_db;
select avg(price) as '商品平均价格' from day06_db.shop_db;
select max(price) as '商品最大价格' from day06_db.shop_db;

# 5.分组查询 group by 列 先分组，再聚合
select apartment ,count(*) as '部门人数' from day06_db.user group by apartment;
select apartment ,avg(salary) as '部门平均薪资' from day06_db.user group by apartment;
select apartment ,max(salary) as '部门最大薪资' from day06_db.user group by apartment;
select apartment ,min(salary) as '部门最小薪资' from day06_db.user group by apartment;
select apartment ,avg(age) as '部门平均年龄' from day06_db.user group by apartment;
# where 解决分组前的条件，后边不能使用聚合函数
# having 解决分组后的条件，后边可以使用聚合函数
select apartment as '部门名称',
       count(*) as '部门人数'
from day06_db.user group by apartment having count(*)>3;

# 6.分页查询 limit 起始索引，每页显示的条数
# 起始索引=(当前页-1)*每页显示的条数
select * from day06_db.user limit 0,3;
select * from day06_db.user limit 3,3;

# 7.多表查询
# 7.1 创建主表
create table day06_db.category(
    id int primary key auto_increment,
    name varchar(20) not null unique
);
insert into day06_db.category values (null,'手机'), (null,'电脑');
# 7.2 创建从表
# 设置外键约束
create table day06_db.product(
    id int primary key auto_increment,
    name varchar(20) not null unique,
    price decimal(7,2) not null,
    category_id int not null,
    # 在公共区域中定义外键约束
    foreign key (category_id) references day06_db.category(id)
);
insert into day06_db.product values
                    (null,'手机1',1000,1),
                    (null,'手机2',2000,1),
                    (null,'手机3',3000,1),
                    (null,'电脑1',5000,2),
                    (null,'电脑2',6000,2),
                    (null,'电脑3',7000,2);

# 8.无条件查询
select * from day06_db.category,day06_db.product;
select * from day06_db.category join day06_db.product;

# 9.有条件查询
# 9.1 内连接
select * from day06_db.category c join day06_db.product p on c.id = p.category_id;

# 9.2 左外连接
# left join 左侧的内容全要，右侧的内容有则要，无则不

select * from day06_db.category c left join day06_db.product p on c.id = p.category_id;
# 9.3 右外连接
# right join 右侧的内容全要，右侧的内容有则要，无则不
select * from day06_db.product p right join day06_db.category c on c.id = p.category_id;
```

### 1. 全外连接查询

```python
# union 去重
select * from day06_db.category c left join day06_db.product p on c.id = p.category_id
union
select * from day06_db.category c right join day06_db.product p on c.id = p.category_id;
# union all 不能去重
select * from day06_db.category c left join day06_db.product p on c.id = p.category_id
union all
select * from day06_db.category c right join day06_db.product p on c.id = p.category_id;
```

### 3. 子查询

```python
# 查询平均年龄
select avg(age) from day06_db.user;
# 查询年龄大于平均年龄的所有用户
select * from day06_db.user where age > (select avg(age) from day06_db.user);
# 查询员工薪资与平均薪资的差
select username,salary-(select avg(salary) from day06_db.user) as '薪资差' from day06_db.user;
```

## 四、pymysql

### 1.查询数据

```python
# 导入pymysql
import pymysql
# 创建mysql连接
conn = pymysql.connect(
         host='192.168.14.21',
         port=3306,
         user='root',
         password='123456'
		 charset='utf8')
# 创建游标
cursor = conn.cursor()
# 执行sql
res = cursor.execute("""show databases""")
res = cursor.execute('select * from second_db.game_table;')
# 查看受影响的行数
print(res)
# 查询的结果 fetchone() 查看一条 返回元组
for i in range(res):
    print(f'查询到的第{i}条数据为：{cursor.fetchone()}')
# 查询的结果 fetchall() 查看全部 返回二维元组
print(f'查询到的所有数据为：{cursor.fetchall()}')
# 关闭游标
cursor.close()
# 关闭连接
conn.close()
```

### 2. 修改数据

```python
# 导入pymysql
import pymysql

# 创建mysql连接
conn = pymysql.connect(
    host='192.168.14.21',
    port=3306,
    user='root',
    password='123456',
    charset='utf8')
# 创建游标
cursor = conn.cursor()
# 执行sql
try:
    # 小弟干活
    res = cursor.execute("""
    insert into second_db.user values (2,'老潘','上午迟到表演节目')
    """)
    # 大哥确认
    conn.commit()
except Exception as e:
    print(f'小弟出错：{e}')
    # 小弟出错，大哥回滚撤销
    conn.rollback()
# 关闭游标
cursor.close()
# 关闭连接
conn.close()
```
