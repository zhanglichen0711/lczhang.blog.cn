## redis(键值型数据库)

概念：用空间换时间，用内存换速度

### 1. redis连接

```python
# 导包
import redis
# 创建redis连接
r = redis.Redis(
    # host='localhost',默认值可以省略
    # port=6379,默认值可以省略
    # db=0,默认值可以省略
    decode_responses=True)
# 测试连接
try:
    res = r.ping()
    print(f'连接成功：{res}')
except Exception as e:
    print(f'连接失败：{e}')
# 关闭连接
r.close()
```

### 2. 字符串操作

```python
import redis
r = redis.Redis(decode_responses=True)
# 设置单个键值对
r.set('name','小王')
# 取出单个值
print(r.get('name'))
# 设置多个值
r.mset({'name':'小张','age':18})
# 取出多个值
print(r.mget('name','age'))
# 设置过期时间
# 位置传参
r.setex('gender',10,'小李')
# 关键字传参
r.setex(name='李正洋',value=25,time=10)
r.close()
```

### 3. 数值操作

```python
import redis
r = redis.Redis(decode_responses=True)
# 设置初始值
r.set('count','0')
# 自增
r.incr('count')
print(r.get('count'))
# 自减
r.decr('count')
print(r.get('count'))
# 自增指定值
r.incrby('count',5)
print(r.get('count'))
# 自减指定值
r.decrby('count',5)
print(r.get('count'))
r.close()
```

### 4. 哈希Hash操作

```python
import redis
r = redis.Redis(decode_responses=True)
# 设置字典类型的值
# r.hset(resdis的键,值的键,值的值)
r.hset('user', 'name', '张三')
r.hset('user', 'age', '18')
# 获取字典类型的某个值
print(r.hget('user', 'name'))
# 获取字典类型的所有值(某个字典)
print(r.hgetall('user'))
# 设置多个键值对
# r.hmset(redis的键,{键:值,键:值})
r.hmset('user', {'gender': '男', 'money': '0'})
# 取出多个值
# r.hmget(redis的键,['键','键'])
print(r.hmget('user', ['name','age','gender', 'money']))
# 取出字典的所有键
print(r.hkeys('user'))
# 取出字典的所有值
print(r.hvals('user'))
# 删除对应字典的属性(可以删多个)
# r.hdel(redis的键,值的键,值的键...)
r.hdel('user', 'money', 'age')
r.close()
```

### 5. 列表操作

```python
import redis

r=redis.Redis(decode_responses=True)
# 1.左侧添加 lpush 从右往左添加
r.lpush('list', '1', '2')
# 2.右侧添加 rpush 从左往右添加
r.rpush('list', '3', '4')
# 3.左侧删除 lpop
r.lpop('list')
# 4.右侧删除 rpop
r.rpop('list')
# 5.查看列表长度
print(r.llen('list'))
# 6.查看列表所有元素
print(r.lrange('list', 0, -1))
```

### 6. 集合操作

```python
import redis
r=redis.Redis(decode_responses=True)
# 添加元素
r.sadd('set', '1', '2', '3', '4')
# 查看所有元素
print(r.smembers('set'))
# 判断元素是否存在
print(r.sismember('set', '1'))
# 获取集合元素的数量
print(r.scard('set'))
# 随机删除一个元素
print(r.spop('set'))
print(r.smembers('set'))
# 指定删除一个元素
r.srem('set', '1')
print(r.smembers('set'))
# 交集 intersection
r.sadd('s1', '1', '2', '3', '4', '5')
r.sadd('s2', '2', '3', '4', '5', '6')
print(r.sinter('s1', 's2'))
# 并集 union
print(r.sunion('s1', 's2'))
# 差集 difference a有b没有
print(r.sdiff('s1', 's2'))
```

### 7. 有序集合操作

```python
import redis

r=redis.Redis(decode_responses=True)
# 有序集合
r.zadd('userlist', {'张三': 88,
       '李四':90,
       '王五':95,
       '赵六':85})
# 按照分数升序排序
print(r.zrange('userlist', 0, -1,withscores= True))
# 按照分数降序排序
print(r.zrevrange('userlist', 0, -1,withscores= True))
# 获取指定元素
print(r.zscore('userlist', '张三'))
# 获取指定区间的元素
print(r.zrangebysc1ore('userlist', 85, 95))
# 增加元素的分数
r.zincrby('userlist', 10, '张三')
r.zincrby('userlist', -10, '王五')
# 获取排名
print(r.zrank('userlist', '张三'))
```
